84int32_t uFsm::hash( uint32_t sourceState, uint32_t event )
85{
86 int32_t i=0;
87 int32_t where = ( ( event << 8 ) + sourceState) % maxNumTransitions_;
88
89 // Look for the first available position

145bool uFsm::control( uint32_t event, void *parameters )
146{
147 bool result = false;
148 int32_t transition_num;
149
150 // Insert the received event in the insertInQueue
151 insertInQueue(event, parameters);
152
153 while( !Evlst_.empty() )
154 {
155 uFsmEvent* pe = Evlst_.front();
156 assert( pe!=NULL );
157 // check if it is valid transition
158 if ( (transition_num = hash(pe->id_)) >= 0 )
159 {
160 if ( arrayTrans_[transition_num].index == -1 )
161 {
162 // Missing transition - generate an internal error event
163 generateEvent(INTERNAL_ERROR);
164 }
165 else
166 {
167 // Execute an event handler
168 result = (handler_->*handler_->functions_[
169 arrayTrans_[transition_num].index])(pe->parameters_);
170
171 // Change the FSM's state
172 currentState_ = arrayTrans_[transition_num].destination_state;
173 }
174 }
175
176 // Missing transition - generate an internal error event
177 else
178 {
179 assert( pe->id_ == INTERNAL_ERROR );
180 generateEvent(INTERNAL_ERROR);
181 }
182
183 Evlst_.pop_front();
184 delete pe;
185 }
186
187 return result;
188}
